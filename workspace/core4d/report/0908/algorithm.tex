% ============================================================================
%  Algorithm blocks for the human-object retargeting method.
%  Two contributions are named in the captions and marked in-line:
%    (C1) Multi-Scale Contact Reward
%    (C2) Part-wise Penetration Constraints
%  Pseudocode style follows standard sampling-based MPC / CEM.
%
%  Required packages (add to preamble):
%    \usepackage{algorithm}
%    \usepackage{algpseudocode}   % algorithmicx
%    \usepackage{amsmath,amssymb}
%
%  Notation
%    x^ref_{0:T} : kinematic reference trajectory (qpos, object pose, contact mask m_t)
%    U          : control knots (PD targets); u_{0:T-1} = interp(U, tau)
%    N_W        : number of sampled control sequences
%    Sigma^i    : annealed sampling covariance (broad->narrow over iterations & horizon)
%    phi_P(x)   : signed distance of robot part-group P to the object surface (object-local SDF)
%    p^e, p^{*,e}: end-effector e contact point and its object-attached anchor point
%    m^e_t      : reference contact mask for end-effector e at time t
%    B          : thin surface band [d_lo, d_hi] = [-1mm, +3mm]
%    theta_P    : per-part signed-clearance threshold (encodes the part's contact role)
% ============================================================================

% ---------------------------------------------------------------------------
% ALGORITHM 1 : outer receding-horizon sampling MPC (backbone) that uses the
%               two contributions via the cost/selection subroutine (Alg. 2).
% ---------------------------------------------------------------------------
\begin{algorithm}[t]
\caption{Sampling-Based Retargeting with Multi-Scale Contact Reward
         and Part-wise Penetration Constraints}
\label{alg:core4d-mpc}
\begin{algorithmic}[1]
\Require reference $x^{\mathrm{ref}}_{0:T}$ (qpos, object pose, contact mask $m$),
         horizon $H$, samples $N_W$, iters $N$, elite frac $\rho$,
         min-valid frac $f$, temperature $\lambda$, anneal $\beta_1,\beta_2$, tol $\epsilon$
\State initialise control knots $U \gets \mathrm{interp}^{-1}(u^{\mathrm{ref}})$
\For{each simulation step $s = 0,1,\dots$}          \Comment{outer receding horizon}
  \State $J_{\text{best}} \gets \infty,\; i \gets 0$
  \While{$i < N$ \textbf{and} not converged}         \Comment{inner CEM}
    \State $i \gets i+1$;\; update $\Sigma^{i}_{h}$ by anneal schedule ($\beta_1,\beta_2$)
    \For{$j = 1,\dots,N_W$}
      \State $[W]_j \sim \mathcal N(0,\Sigma^{i})$;\;\; $[U^i]_j \gets U^i + [W]_j$
      \State roll out $x^{j}_{0:H}$ in the physics simulator from $u^j = \mathrm{interp}([U^i]_j,\tau)$
      \State $J^j,\; \mathbf g^j \gets \Call{CostAndGate}{x^{j}_{0:H}, x^{\mathrm{ref}}}$
             \Comment{Alg.~\ref{alg:cost-gate}}
    \EndFor
    \State $U^{i+1} \gets \Call{GatedElite}{\{J^j\},\{\mathbf g^j\},\{[U^i]_j\},\rho,f,\lambda}$
           \Comment{Alg.~\ref{alg:cost-gate}}
    \If{$|J_{\text{best}} - \min_j J^j| < \epsilon$} \State \textbf{break} \Comment{early stop}
    \EndIf
    \State $J_{\text{best}} \gets \min_j J^j$
  \EndWhile
  \State execute first $n_{\text{exec}}$ controls of $U$;\; advance simulator
\EndFor
\State \Return refined dynamically-feasible controls $U^{*}$
\end{algorithmic}
\end{algorithm}

% ---------------------------------------------------------------------------
% ALGORITHM 2 : the two contributions.
%   CostAndGate : per-sample cost = tracking + (C1) Multi-Scale Contact Reward
%                 + (C2) Part-wise Penetration Constraints (soft layer), plus
%                 the per-part gate statistics (C2 hard layer).
%   GatedElite  : constraint-gated elite selection with least-violation fallback.
% ---------------------------------------------------------------------------
\begin{algorithm}[t]
\caption{Multi-Scale Contact Reward and Part-wise Penetration Constraints,
         with Gated-Elite Selection}
\label{alg:cost-gate}
\begin{algorithmic}[1]
\Function{CostAndGate}{$x_{0:H}, x^{\mathrm{ref}}$}
  \State $J \gets 0$
  \For{$t = 0,\dots,H$}
    \State $J \mathrel{+}= \underbrace{\big\| (q_t - q^{\mathrm{ref}}_t)\,W_{\text{loc}} \big\|}_{\text{local-frame tracking}}
                         + \underbrace{\alpha_o\,\|\,o_t - o^{\mathrm{ref}}_t\|}_{\text{object pose}}$
    \For{each end-effector $e$ with contact $m^e_t=1$}
             \Comment{\textbf{C1: Multi-Scale Contact Reward}}
      \State $r^{\mathrm{coarse}}_{t,e} \gets g\,\exp\!\big(-\|p^e_t - p^{\star,e}_t\|/\sigma_c\big)$
             \Comment{broad scale ($\sigma_c$)}
      \State $r^{\mathrm{fine}}_{t,e} \gets s\,\exp\!\big(-|\phi_{e}(x_t)|/\sigma_f\big)\cdot \mathbf 1[\phi_e(x_t)\in B]$
             \Comment{sharp scale ($\sigma_f$)}
      \State $J \mathrel{-}= r^{\mathrm{coarse}}_{t,e} + r^{\mathrm{fine}}_{t,e}$
    \EndFor
    \For{each part group $P \in \{\text{hand},\text{body},\text{leg}\}$}
             \Comment{\textbf{C2: Part-wise Penetration Constraints}, soft}
      \State $J \mathrel{+}= w_P\,\max\!\big(0,\; \delta_P - \phi_P(x_t)\big)$
    \EndFor
    \State $J \mathrel{+}= $ ground-penetration terms (body / foot height $z$)
  \EndFor
  \For{each part group $P$}
             \Comment{hard}
    \State $g^{\min}_P \gets \min_t \phi_P(x_t)$;\quad
           $g^{\text{vio}}_P \gets \tfrac1H\sum_t \mathbf 1[\phi_P(x_t) < \theta_P]$
  \EndFor
  \State \Return $J,\; \mathbf g = \{(g^{\min}_P, g^{\text{vio}}_P)\}_P$
\EndFunction
\Statex
\Function{GatedElite}{$\{J^j\},\{\mathbf g^j\},\{U_j\},\rho,f,\lambda$}
  \State valid$_j \gets \bigwedge_P \big[\, g^{\min,j}_P \ge \text{floor}_P \;\wedge\; g^{\text{vio},j}_P \le \tau_P \,\big]$
         \Comment{per-part AND}
  \If{$\#\{j:\text{valid}_j\} \ge f\,N_W$}
    \State $\mathcal E \gets$ top-$\lceil\rho N_W\rceil$ of $\{j:\text{valid}_j\}$ by lowest $J^j$
  \Else                                                   \Comment{least-violation fallback}
    \State $\mathcal E \gets$ top-$\lceil\rho N_W\rceil$ by lowest
           $\big(\text{depth}^j + \text{vio}^j - 10^{-3}\,\tilde J^j\big)$
  \EndIf
  \State $w_j \gets \dfrac{\exp(-J^j/\lambda)\,\mathbf 1[j\in\mathcal E]}{\sum_{k\in\mathcal E}\exp(-J^k/\lambda)}$
  \State \Return $\sum_j w_j\,U_j$                        \Comment{softmax elite refit}
\EndFunction
\end{algorithmic}
\end{algorithm}
